SELECT DISTINCT
    EMPLOYEEID,
    GENDER AS NEW_GENDER,
    DEPARTMENT AS NEW_DEPARTMENT,
    HIREDATE AS NEW_HIREDATE,
    SALARY AS OLD_SALARY,
    BONUS AS OLD_BONUS,
    SALARY + BONUS AS NEW_SALARY_INCREMENT,
    SALARY - BONUS AS NEW_SALARY_DECEREMENT,
    SALARY * 2 AS NEW_BIG_SALARY,
    SALARY / 10 AS NEW_DIVIDED_SALARY,
    SALARY % 1000 AS NEW_MODULES
FROM EMP_DATA
WHERE Department = 'HR'
  AND HireDate >= '2025-12-01'
  AND Salary >= 100000
  AND Bonus <= 20000
  AND Gender = 'M'
  AND HireDate <> '2025-12-14'
  AND HireDate != '2025-12-10'
ORDER BY Salary DESC;
